package com.plamdi.mbaya.ilesetponts;

import android.annotation.SuppressLint;
import android.app.Activity;
import android.os.Bundle;
import android.webkit.WebChromeClient;
import android.webkit.WebSettings;
import android.webkit.WebView;
import android.webkit.WebViewClient;

/**
 * Affiche le jeu (assets/index.html) dans une WebView plein écran.
 * Tout est local : le jeu fonctionne sans connexion Internet.
 */
public class MainActivity extends Activity {

    private WebView web;

    @SuppressLint("SetJavaScriptEnabled")
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        web = new WebView(this);
        WebSettings s = web.getSettings();
        s.setJavaScriptEnabled(true);      // le jeu est écrit en JavaScript
        s.setDomStorageEnabled(true);      // sauvegarde de la progression (localStorage)
        s.setAllowFileAccess(true);
        s.setMediaPlaybackRequiresUserGesture(false);

        // Reste dans l'application : aucun lien ne s'ouvre dans un navigateur externe
        web.setWebViewClient(new WebViewClient());
        web.setWebChromeClient(new WebChromeClient());
        web.setBackgroundColor(0xFF0A4C8C);

        setContentView(web);
        web.loadUrl("file:///android_asset/index.html");
    }

    @Override
    protected void onPause() {
        super.onPause();
        web.onPause();
    }

    @Override
    protected void onResume() {
        super.onResume();
        web.onResume();
    }

    @Override
    protected void onDestroy() {
        web.destroy();
        super.onDestroy();
    }
}

NOM A CREER : .github/workflows/build-apk.yml
==================================================
# Compile l'APK dans le cloud (GitHub), sans ordinateur installé.
# 1. Mets ce dossier IlesEtPonts dans un dépôt GitHub (privé ou public).
# 2. Onglet « Actions » > « Build APK » > « Run workflow ».
# 3. Quand c'est fini, télécharge l'artefact « IlesEtPonts-apk » : il contient app-debug.apk.
name: Build APK

on:
  workflow_dispatch:
  push:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: android-actions/setup-android@v3

      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'

      - name: Télécharger Three.js dans les assets
        run: curl -fsSL -o app/src/main/assets/three.min.js https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js

      - name: Supprimer les fichiers mal nommés
        run: find app -name '* *' -delete

      - name: Compiler l'APK de test
        run: gradle assembleDebug --no-daemon

      - uses: actions/upload-artifact@v4
        with:
          name: IlesEtPonts-apk
          path: app/build/outputs/apk/debug/app-debug.apk
